C# / Unity 

using UnityEngine;

public class noSelfColl : MonoBehaviour
{
    void Start()
    {
        Collider2D[] colliders = transform.GetComponentsInChildren();
        for (int i = 0; i < colliders.Length; i++)
        {
            for (int k = i + 1; k < colliders.Length; k++)
            {
                Physics2D.IgnoreCollision(colliders[i], colliders[k]);
            }
        }
    }
}
